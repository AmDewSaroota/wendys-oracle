0 ต้นฉบับ — AE Expression ทำงานอย่างไร

ก่อนจะแปลงมา Unity ทำความเข้าใจว่าสคริปต์ AE ต้นฉบับมีหลักการอะไรบ้าง:

① คำนวณหน่วงเวลาจาก Layer Index

Delay             = 0.15   // วินาที ต่อ 1 ช่วง
Follower_Multiplier = index - L.index
Total_Delay         = Follower_Multiplier * Delay

ใน AE ทุก layer มีหมายเลข index อัตโนมัติ — ผีเสื้อตัวที่ 1 index = 2, ตัวที่ 2 index = 3 และต่อไปเรื่อยๆ (leader = 1)
ผลคือ ยิ่งตัวไหนอยู่ไกลจาก leader ในลำดับ layer → หน่วงเวลานานขึ้นตามสัดส่วน

② ชดเชยตำแหน่งด้วยความเร็ว (Velocity Compensation)

P_Current = L.transform.position.value         // ตำแหน่ง leader ตอนนี้
V_Current = L.transform.position.velocityAtTime(time)  // ความเร็ว leader ตอนนี้

P_Lag = P_Current - V_Current * Total_Delay    // ถอยหลังตามทิศที่บิน

แนวคิดคือ: "ถ้า leader กำลังบินไปทางไหน follower ควรอยู่ด้านหลังตรงนั้น"
สูตรนี้ประมาณตำแหน่งย้อนกลับไปตามเวกเตอร์ความเร็ว — ง่ายและเร็ว แต่เป็นการ approximation (ไม่ใช่ path จริง)

ทำไมถึง approximate?
AE ไม่มี position history — มันรู้แค่ "ตอนนี้ velocity เท่าไหร่" จึงคาดเดาตำแหน่งในอดีตด้วยสูตรเส้นตรง
ถ้า leader หักเลี้ยวกะทันหัน follower จะยังคงชดเชยตาม velocity เก่า ซึ่งพอดีกับการบินของผีเสื้อ (ดูเป็นธรรมชาติ)

③ เพิ่ม Wiggle — การส่ายตัวอิสระ

wiggle_frequency = 0.3   // ช้า = ส่ายนุ่มนวล, เร็ว = สั่นกระวนกระวาย
wiggle_amplitude = 500   // ระยะการกระจายตัว (หน่วย pixel AE)

wiggle_X = wiggle(wiggle_frequency, wiggle_amplitude)[0]
wiggle_Y = wiggle(wiggle_frequency, wiggle_amplitude)[1]
wiggle_Z = wiggle(wiggle_frequency, wiggle_amplitude)[2]

AE wiggle() สร้าง smooth random noise ด้วย Perlin-like algorithm — แต่ละ layer ได้ค่า noise ต่างกันโดยอัตโนมัติ (AE จัดการ seed ให้เอง)

④ รวมผล

1 Velocity-Lag — ผีเสื้อตามกัน

AE ใช้วิธีชดเชยตำแหน่งด้วยความเร็วปัจจุบัน ซึ่งเป็นการ ประมาณ ว่า follower อยู่ "ด้านหลัง" leader แค่ไหน Unity ทำได้ 2 วิธี:

วิธี A
Velocity Lag
เหมือน AE เลย — ง่าย เขียนน้อย
Vector3 lagPos =
  leader.position
  - leaderVelocity * totalDelay;

transform.position =
  lagPos + wiggleOffset;
วิธี B
History Buffer แนะนำ
แม่นกว่า — follower โค้งตามเส้นทางจริง
// เก็บ history ทุก frame
positionHistory.Enqueue(
  leader.position
);

// follower อ่าน N frames ที่แล้ว
transform.position =
  positionHistory[followerIndex];
เมื่อไหร่ใช้วิธีไหน?
วิธี A — prototype เร็ว หรือ path ของ leader เป็นเส้นตรง
วิธี B — path โค้ง เช่น Sine Wave หรือ Spline → follower จะโค้งตามสวยกว่ามาก

2 Wiggle → Perlin Noise

AE wiggle(freq, amp) ใช้ noise แบบ smooth random — ใน Unity ใช้ Mathf.PerlinNoise แทนได้ตรงๆ

float t = Time.time * wiggleFrequency;

float x = (Mathf.PerlinNoise(t + seed, 0f) - 0.5f) * 2f * wiggleAmplitude;
float y = (Mathf.PerlinNoise(0f, t + seed) - 0.5f) * 2f * wiggleAmplitude;
float z = (Mathf.PerlinNoise(t + seed, t + seed) - 0.5f) * 2f * wiggleAmplitude;

Vector3 wiggleOffset = new Vector3(x, y, z);
⚠️ ต้องใช้ seed ต่างกันทุกตัว
ถ้าทุกตัวใช้ seed เดียวกัน → wiggle พร้อมกันหมด ดูไม่เป็นธรรมชาติ
แก้ง่ายๆ: float seed = followerIndex * 100f;

3 Mapping — AE → Unity

AE Expression Unity (C#) หมายเหตุ
layer("Butterfly_Leader") GameObject leaderObj assign ใน Inspector
index - L.index followerIndex array index ของ follower
.velocityAtTime(time) (pos - prevPos) / Time.deltaTime คำนวณเองทุก frame
wiggle(freq, amp) Mathf.PerlinNoise ต้องใช้ seed แยกแต่ละตัว
Comp space (px) World space (units) scale ลงเยอะ — ดูหมายเหตุด้านล่าง
⚠️ เรื่อง Scale
AE ทำงานใน 2D composition space หน่วยเป็น pixel — Unity เป็น 3D world units
wiggle_amplitude = 500 ใน AE อาจต้อง scale เหลือ 0.5f ใน Unity
ขึ้นอยู่กับขนาด scene ของน้องค่ะ — ปรับ trial and error

4 โครงสร้าง Script คร่าวๆ

ถ้าน้องจะ implement วิธี A (ง่ายสุด) — โครงสร้าง C# จะประมาณนี้:

public class ButterflyFollower : MonoBehaviour
{
    public Transform leader;
    public int followerIndex;
    public float delay = 0.15f;
    public float wiggleFreq = 0.3f;
    public float wiggleAmp = 0.5f;

    private Vector3 prevLeaderPos;
    private float seed;

    void Start()
    {
        prevLeaderPos = leader.position;
        seed = followerIndex * 100f;
    }

    void Update()
    {
        // คำนวณ velocity ของ leader
        Vector3 vel = (leader.position - prevLeaderPos) / Time.deltaTime;
        prevLeaderPos = leader.position;

        // lag position
        float totalDelay = followerIndex * delay;
        Vector3 lagPos = leader.position - vel * totalDelay;

        // wiggle
        float t = Time.time * wiggleFreq;
        Vector3 w = new Vector3(
            (Mathf.PerlinNoise(t + seed, 0f) - .5f) * 2f * wiggleAmp,
            (Mathf.PerlinNoise(0f, t + seed) - .5f) * 2f * wiggleAmp,
            (Mathf.PerlinNoise(t + seed, t + seed) - .5f) * 2f * wiggleAmp
        );

        transform.position = lagPos + w;
    }
}
วิธีใช้: Attach script นี้ให้กับ butterfly แต่ละตัว → กำหนด leader + followerIndex ใน Inspector
ตัวแรก followerIndex = 1, ตัวสอง = 2 และต่อไปเรื่อยๆ

★ Hero Butterfly — งานปัจจุบัน

ใช้อันนี้

ไม่ใช่ฝูง ไม่มี leader-follower — แต่ละตัวบินอิสระ มี boundary, มีอายุ, fade ออกเองเมื่อหมดเวลา

หลักการ: Spawn → บินด้วย Perlin noise steering → ผลักกลับเมื่อใกล้ขอบ → fade + destroy เมื่อหมดอายุ

ButterflyHero.cs

using UnityEngine;

public class ButterflyHero : MonoBehaviour
{
    [Header("Flight")]
    public float speed         = 1.5f;
    public float turnSmooth    = 1.5f;   // สูง = หักเลี้ยวเร็ว
    public float noiseFrequency = 0.35f;

    [Header("Boundary")]
    public Vector3 boundsMin = new Vector3(-8f, 0.5f, -4f);
    public Vector3 boundsMax = new Vector3( 8f, 5.0f,  4f);
    public float  boundaryPush = 2.5f;  // ระยะเริ่มผลักก่อนถึงขอบ

    [Header("Lifetime")]
    public float lifetime    = 12f;
    public float fadeOutTime = 2f;

    private float    seed;
    private Vector3  velocity;
    private Renderer rend;
    private float    age;

    void Start()
    {
        seed     = Random.value * 1000f;
        rend     = GetComponentInChildren<Renderer>();
        velocity = Random.onUnitSphere * speed;  // ทิศเริ่มต้นสุ่ม
    }

    void Update()
    {
        age += Time.deltaTime;

        // 1. Noise steering — ทิศทางบินแบบ organic
        float t = Time.time * noiseFrequency + seed;
        Vector3 noiseDir = new Vector3(
            Mathf.PerlinNoise(t,        0f       ) - 0.5f,
            Mathf.PerlinNoise(0f,       t        ) - 0.5f,
            Mathf.PerlinNoise(t * 0.7f, t * 1.3f) - 0.5f
        );

        // 2. Boundary push — ผลักกลับเมื่อใกล้ขอบ
        Vector3 pos  = transform.position;
        Vector3 push = Vector3.zero;
        float   m    = boundaryPush;

        push.x += Mathf.InverseLerp(boundsMin.x + m, boundsMin.x, pos.x);
        push.x -= Mathf.InverseLerp(boundsMax.x - m, boundsMax.x, pos.x);
        push.y += Mathf.InverseLerp(boundsMin.y + m, boundsMin.y, pos.y);
        push.y -= Mathf.InverseLerp(boundsMax.y - m, boundsMax.y, pos.y);
        push.z += Mathf.InverseLerp(boundsMin.z + m, boundsMin.z, pos.z);
        push.z -= Mathf.InverseLerp(boundsMax.z - m, boundsMax.z, pos.z);

        // 3. รวมทิศ → smooth velocity
        Vector3 desired = (noiseDir.normalized + push * 2f).normalized * speed;
        velocity = Vector3.Lerp(velocity, desired, turnSmooth * Time.deltaTime);
        transform.position += velocity * Time.deltaTime;

        // 4. หันหน้าตามทิศบิน
        if (velocity.sqrMagnitude > 0.01f)
            transform.rotation = Quaternion.Slerp(
                transform.rotation,
                Quaternion.LookRotation(velocity),
                turnSmooth * Time.deltaTime
            );

        // 5. Fade out → destroy
        HandleLifetime();
    }

    void HandleLifetime()
    {
        if (age < lifetime - fadeOutTime) return;

        float alpha = 1f - (age - (lifetime - fadeOutTime)) / fadeOutTime;
        Color c = rend.material.color;
        c.a = Mathf.Max(0f, alpha);
        rend.material.color = c;

        if (age >= lifetime)
            Destroy(gameObject);
    }
}
⚠️ Fade ใช้ได้เฉพาะ material ที่ตั้ง Rendering Mode = Fade / Transparent
URP → ตั้ง Surface Type = Transparent ใน material ด้วยค่ะ

Parameter ที่ปรับบ่อย

Parameter Default ปรับเพื่อ
speed 1.5f บินเร็ว/ช้า — 0.8–2.5 เหมาะกับ magical scene
turnSmooth 1.5f ต่ำ = เลี้ยวนุ่ม, สูง = หักเลี้ยวฉับพลัน
noiseFrequency 0.35f ต่ำ = path ยาว ช้า, สูง = วนสั้น ถี่
boundaryPush 2.5f ระยะเริ่มโค้งกลับ — ยิ่งสูงยิ่ง smooth
lifetime 12f อายุรวม (วินาที) — ปรับตาม UX ที่ต้องการ
fadeOutTime 2f ช่วง fade ก่อน destroy
